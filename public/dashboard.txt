    <script>
        let myChartInstance = null;
        let studentProfileData = {
            name: "Rahul Sharma",
            classLevel: "Class 10th Math",
            email: "rahul.math@amit.edu",
            phone: "+91 98765 43210",
            photo: "https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=150"
        };

        async function fetchDashboardData() {
            try {
                const studentId = "AMIT_7821";

                const response = await fetch(`http://localhost:8080/api/analytics/${studentId}`);
                const result = await response.json();

                if (result.success) {
                    const data = result.data;
                    
                    document.getElementById('lblAccuracy').textContent = data.overallAccuracy + '%';
                    document.getElementById('lblSpeed').textContent = data.averageSpeedPerQuestion + 's';
                    document.getElementById('lblAttempted').textContent = data.totalQuestionsAttempted;

                    // 1. Render AI Insights
                    const insightsContainer = document.getElementById('insightsContainer');
                    insightsContainer.innerHTML = '';
                    data.aiInsights.forEach(insight => {
                        insightsContainer.innerHTML += `<div class="insight-box">${insight}</div>`;
                    });

                    // 2. Render Topic Metrics
                    const topicsContainer = document.getElementById('topicsContainer');
                    topicsContainer.innerHTML = '';
                    data.topicMetrics.forEach(topic => {
                        let accuracy = Math.round((topic.correct / topic.attempted) * 100) || 0;
                        topicsContainer.innerHTML += `
                            <div class="topic-row">
                                <span class="topic-name"><i class="ph-fill ph-triangle" style="font-size: 10px; color: var(--accent);"></i> ${topic.topicName}</span>
                                <span class="topic-stats">Att: ${topic.attempted} | Corr: ${topic.correct} | <strong style="color: var(--success);">${accuracy}%</strong></span>
                            </div>
                        `;
                    });

                    // 3. Render Daily Quiz History Logs
                    const quizHistoryContainer = document.getElementById('quizHistoryContainer');
                    quizHistoryContainer.innerHTML = '';
                    
                    const quizLogs = [
                        { date: '29 July 2026', title: 'Calculus & Limits', score: '9/10 (90%)' },
                        { date: '28 July 2026', title: 'Algebraic Identities', score: '8/10 (80%)' },
                        { date: '27 July 2026', title: 'Trigonometric Ratios', score: '10/10 (100%)' },
                        { date: '25 July 2026', title: 'Coordinate Geometry', score: '17/20 (85%)' }
                    ];

                    quizLogs.forEach(log => {
                        quizHistoryContainer.innerHTML += `
                            <div class="quiz-log-item">
                                <span class="quiz-date"><i class="ph-fill ph-hash"></i> ${log.date} - <strong style="color: var(--text-main);">${log.title}</strong></span>
                                <span class="quiz-score">${log.score}</span>
                            </div>
                        `;
                    });

                    // 4. Render Interactive Chart.js Graph
                    renderLearningGraph(data.learningCurve);
                }
            } catch (error) {
                console.error("Failed to load dashboard data:", error);
            }
        }

        // Profile Modal Actions
        function openProfileModal() {
            document.getElementById('modalStudentName').textContent = studentProfileData.name;
            document.getElementById('modalClass').textContent = studentProfileData.classLevel;
            document.getElementById('modalEmail').textContent = studentProfileData.email;
            document.getElementById('modalPhone').textContent = studentProfileData.phone;
            document.getElementById('modalStudentImg').src = studentProfileData.photo;
            document.getElementById('profileModal').style.display = 'flex';
        }

        function closeProfileModal() {
            document.getElementById('profileModal').style.display = 'none';
        }

        function renderLearningGraph(curveData) {
            const ctx = document.getElementById('learningCurveChart').getContext('2d');
            
            const labels = curveData.map(item => item.date);
            const accuracies = curveData.map(item => item.accuracy);

            if (myChartInstance) {
                myChartInstance.destroy();
            }

            myChartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Accuracy Progression (%)',
                        data: accuracies,
                        borderColor: '#06B6D4',
                        backgroundColor: 'rgba(6, 182, 212, 0.1)',
                        borderWidth: 3,
                        fill: true,
                        tension: 0.4,
                        pointRadius: 6,
                        pointBackgroundColor: '#4F46E5',
                        pointBorderColor: '#06B6D4',
                        pointBorderWidth: 2
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { display: false }
                    },
                    scales: {
                        y: {
                            beginAtZero: true,
                            max: 100,
                            grid: { color: '#334155' },
                            ticks: { color: '#94A3B8' }
                        },
                        x: {
                            grid: { display: false },
                            ticks: { color: '#94A3B8' }
                        }
                    }
                }
            });
        }

        window.onload = function() {
            fetchDashboardData();
        };
    </script>
