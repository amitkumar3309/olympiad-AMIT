import jwt from 'jsonwebtoken';

// 1. ROLE DEFINITIONS (Kis user ko kya power deni hai)
export enum UserRole {
  STUDENT = 'STUDENT',
  PARENT = 'PARENT',
  ADMIN = 'ADMIN',
  SUPER_ADMIN = 'SUPER_ADMIN'
}

// 2. JWT SESSION PAYLOAD (Token ke andar kya secure data rahega)
export interface IJwtPayload {
  userId: string;
  role: UserRole;
  schoolCode?: string;
  sessionId: string;
}

// 3. AUTHENTICATION SERVICE (Login, OTP aur Token Generation)
export class AuthService {
  
  /**
   * Step A: Generate Secure OTP for Mobile Verification
   */
  public async generateMobileOTP(mobileNumber: string): Promise<string> {
    // 1. Generate 6-digit random secure OTP
    const otpCode = Math.floor(100000 + Math.random() * 900000).toString();
    
    // 2. Save to Redis/Database with 5-minute expiry
    await this.saveOtpToCache(mobileNumber, otpCode, 300); // 300 seconds (5 mins)
    
    // 3. Trigger external SMS API (e.g., Twilio, MSG91 or Firebase Auth)
    await this.sendSmsApi(mobileNumber, `Your AMIT Olympiad verification code is ${otpCode}. Do not share this with anyone.`);
    
    return "OTP sent successfully";
  }

  /**
   * Step B: Verify OTP and Issue Secure JWT Tokens
   */
  public async verifyOTPAndLogin(mobileNumber: string, otpCode: string, role: UserRole): Promise<any> {
    // 1. Check if OTP is correct and not expired
    const isValid = await this.validateOtpFromCache(mobileNumber, otpCode);
    if (!isValid) {
      throw new Error('❌ Invalid or Expired OTP.');
    }

    // 2. Find User in Database
    const user = await this.findUserByMobile(mobileNumber, role);
    if (!user) {
      throw new Error('User not found. Please register first.');
    }

    // 3. Generate Access Token (1 hour validity)
    const accessToken = jwt.sign(
      { userId: user.id, role: user.role, sessionId: Date.now().toString() },
      process.env.JWT_SECRET_KEY,
      { expiresIn: '1h' }
    );

    // 4. Generate Refresh Token (7 days validity - for keeping user logged in)
    const refreshToken = jwt.sign(
      { userId: user.id },
      process.env.JWT_REFRESH_KEY,
      { expiresIn: '7d' }
    );

    return { accessToken, refreshToken, userProfile: user };
  }
}

/* =========================================================
   ROLE-BASED ACCESS MIDDLEWARE (Security Interceptor)
   ========================================================= */

/**
 * Yeh middleware check karega ki API call karne wala insaan sahi role wala hai ya nahi.
 * Example: Agar student ne Admin wali API call ki, toh use Block kar dega.
 */
export const requireRole = (allowedRoles: UserRole[]) => {
  return (req: any, res: any, next: any) => {
    
    const user = req.user as IJwtPayload; // Injected by previous JWT middleware
    
    if (!user || !allowedRoles.includes(user.role)) {
      // Security Action: Log suspicious activity
      console.warn(`🚩 Unauthorized Access Attempt by User: ${user?.userId}`);
      return res.status(403).json({ 
        error: 'Forbidden', 
        message: 'Aapke paas is page ya data ko access karne ki permission nahi hai.' 
      });
    }
    
    next(); // Access Granted
  };
};

   STEP 4: SMART REGISTRATION & VALIDATION SERVICE
   ========================================================= */

import { StudentRepository } from '../repositories/student.repository';
import { FraudDetectionService } from './fraud-detection.service';
// Zod is a popular TypeScript validation library
import { z } from 'zod'; 

// 1. REGEX RULES (System ko sikhana ki asli naam aur number kaisa hota hai)
const mobileRegex = /^[6-9][0-9]{9}$/; // Starts with 6,7,8,9 and exactly 10 digits
const nameRegex = /^[a-zA-Z\s]+$/;     // Only letters and spaces
const repeatedCharsRegex = /(.)\1{3,}/; // Blocks 4 consecutive same characters (e.g., 'aaaa')

// 2. STRICT VALIDATION SCHEMA (Zod Schema)
export const RegistrationSchema = z.object({
  fullName: z.string()
    .min(3, "Name is too short")
    .regex(nameRegex, "Name can only contain letters and spaces")
    .refine(name => name.trim().split(" ").length >= 2, "Please enter your full name (at least 2 words)")
    .refine(name => !repeatedCharsRegex.test(name), "Invalid name format (too many repeated characters)"),
  
  fatherName: z.string().min(3).regex(nameRegex),
  motherName: z.string().min(3).regex(nameRegex),
  
  mobile: z.string().regex(mobileRegex, "Invalid Indian mobile number"),
  email: z.string().email("Invalid email address").optional(),
  
  dob: z.string().refine(date => !isNaN(Date.parse(date)), "Invalid Date"),
  gender: z.enum(['Male', 'Female']),
  studentClass: z.string().min(1, "Class is required"),
  schoolCode: z.string().min(3, "School Code is required"),
  district: z.string().min(2, "District is required"),
  state: z.string().min(2, "State is required"),
});

// 3. PHOTO VALIDATION SERVICE (Image Quality & Size Check)
export class FileValidationService {
  public validateProfilePhoto(file: any): boolean {
    const allowedMimeTypes = ['image/jpeg', 'image/png', 'image/webp'];
    const maxSizeBytes = 2 * 1024 * 1024; // 2 MB

    if (!file) throw new Error("Photo is compulsory for registration.");
    
    if (!allowedMimeTypes.includes(file.mimetype)) {
      throw new Error("❌ Invalid photo format. Only JPG, PNG, and WEBP are allowed.");
    }
    
    if (file.size > maxSizeBytes) {
      throw new Error("❌ Photo size exceeds 2 MB limit.");
    }

    // Future Ready: Integrate AI API (like AWS Rekognition or Google Vision) to detect if a "Face" is visible in the photo
    return true;
  }
}

// 4. REGISTRATION SERVICE (Main Logic)
export class RegistrationService {
  private studentRepo = new StudentRepository();
  private fileValidator = new FileValidationService();
  private fraudService = new FraudDetectionService();

  public async processNewRegistration(data: any, photoFile: any): Promise<any> {
    
    // Step A: Strictly validate text data
    const validatedData = RegistrationSchema.parse(data);

    // Step B: Validate Photo Upload
    this.fileValidator.validateProfilePhoto(photoFile);

    // Step C: Check for Duplicates in Database
    const existingMobile = await this.studentRepo.findByMobile(validatedData.mobile);
    if (existingMobile) {
      throw new Error("⚠️ This mobile number is already registered.");
    }

    if (validatedData.email) {
      const existingEmail = await this.studentRepo.findByEmail(validatedData.email);
      if (existingEmail) {
        throw new Error("⚠️ This email is already registered.");
      }
    }

    // Step D: AI Fraud Check (Check if IP Address made too many accounts today)
    const isSuspicious = await this.fraudService.checkRegistrationFraud(validatedData.mobile);

    // Step E: Save to Database with 'Pending' Status
    // Admin will manually verify the photo and details before making it 'Verified'
    const newStudent = await this.studentRepo.createStudent({
      ...validatedData,
      photoUrl: 'pending_upload_path', // URL generated after Cloud Storage upload
      registrationStatus: isSuspicious ? 'FlaggedForReview' : 'Pending',
      createdAt: new Date(),
      metrics: { totalXP: 0, currentLevelId: 'level_1', dailyStreak: 0, bestStreak: 0 }
    });

    return {
      success: true,
      message: "Registration successful. Please complete OTP verification.",
      studentId: newStudent.studentId
    };
  }
}

import { QuestionRepository } from '../repositories/question.repository';
import { CloudStorageService } from '../services/cloud-storage.service'; // Firebase/AWS S3 Storage

// 1. ADMIN SERVICE (Admin dwara Question Photo/PDF ke sath upload karna)
export class QuestionAdminService {
  private questionRepo = new QuestionRepository();
  private cloudStorage = new CloudStorageService();

  /**
   * Admin naya question banayega (Text ya Photo/PDF ke sath)
   */
  public async createDailyQuestion(adminId: string, questionData: any, mediaFile?: any): Promise<any> {
    
    // Server-side validation for Answer
    if (!questionData.correctAnswer) {
      throw new Error("Correct Answer is mandatory.");
    }

    let uploadedMediaUrl = null;
    let mediaType = null;

    // Agar Admin ne Image ya PDF upload ki hai
    if (mediaFile) {
      // Step A: Validate file type (Only Image or PDF allowed)
      const allowedTypes = ['image/jpeg', 'image/png', 'image/webp', 'application/pdf'];
      if (!allowedTypes.includes(mediaFile.mimetype)) {
        throw new Error("❌ Invalid file format. Only JPG, PNG, WEBP and PDF are allowed.");
      }

      // Step B: Upload file to Cloud Storage (Firebase Storage / AWS S3)
      uploadedMediaUrl = await this.cloudStorage.uploadQuestionMedia(mediaFile);
      mediaType = mediaFile.mimetype === 'application/pdf' ? 'PDF' : 'IMAGE';
    }

    // Dono mein se ek cheez compulsory hai: Ya toh Text ho, ya Photo/PDF ho
    if (!questionData.questionText && !uploadedMediaUrl) {
      throw new Error("Aapko ya toh Question ka Text likhna hoga, ya uski Photo/PDF upload karni hogi.");
    }

    // Database mein save karna
    const newQuestion = await this.questionRepo.create({
      ...questionData,
      content: {
        questionText: questionData.questionText || "", // Optional if photo is uploaded
        mediaUrl: uploadedMediaUrl,                    // Photo/PDF ka secure link
        mediaType: mediaType,                          // 'IMAGE' ya 'PDF'
        correctAnswer: questionData.correctAnswer,
        explanation: questionData.explanation || ""
      },
      type: 'DailyChallenge',
      isActive: true,
      createdByAdminId: adminId,
      createdAt: new Date(),
    });

    return {
      success: true,
      message: "Daily Challenge Question scheduled successfully.",
      questionId: newQuestion.questionId,
      mediaUrl: uploadedMediaUrl
    };
  }
}

import { StudentRepository } from '../repositories/student.repository';
import { JourneyRepository } from '../repositories/journey.repository';
import { NotificationService } from './notification.service';

// 1. LEVEL THRESHOLDS CONFIGURATION (Level chart mapping)
export const LEVEL_THRESHOLDS = [
  { level: 1, name: "Math Explorer", requiredXP: 0 },
  { level: 2, name: "Number Ninja", requiredXP: 500 },
  { level: 3, name: "Algebra Warrior", requiredXP: 1500 },
  { level: 4, name: "Geometry Master", requiredXP: 3000 },
  { level: 5, name: "Olympiad Legend", requiredXP: 5000 }
];

// 2. XP & LEVEL MANAGEMENT SERVICE
export class XpAndLevelService {
  private studentRepo = new StudentRepository();
  private notificationService = new NotificationService();

  /**
   * Kisi bhi activity (Challenge, Practice, Streak) par bache ko XP dena
   */
  public async awardXP(studentId: string, xpAmount: number, reason: string): Promise<any> {
    
    // 1. Student ki current profile fetch karo
    const student = await this.studentRepo.findById(studentId);
    if (!student) throw new Error("Student not found");

    const oldXP = student.metrics.totalXP;
    const newXP = oldXP + xpAmount;
    
    // 2. Check karna ki kya Naya Level unlock hua hai?
    const oldLevel = this.calculateLevel(oldXP);
    const newLevel = this.calculateLevel(newXP);
    let levelUpOccurred = false;

    // 3. XP aur Level database mein update karo
    await this.studentRepo.updateMetrics(studentId, {
      totalXP: newXP,
      currentLevelId: newLevel.level.toString()
    });

    // 4. Log the XP history (for Analytics and Audit)
    await this.studentRepo.logXpHistory(studentId, xpAmount, reason, new Date());

    // 5. Agar Level Up hua hai, toh bache ko system Notification bhejo!
    if (newLevel.level > oldLevel.level) {
      levelUpOccurred = true;
      await this.notificationService.sendSystemNotification(
        studentId,
        "🎉 LEVEL UP!",
        `Congratulations! You have reached Level ${newLevel.level} and become a ${newLevel.name}.`
      );
    }

    return {
      success: true,
      addedXP: xpAmount,
      totalXP: newXP,
      levelUpOccurred,
      currentLevel: newLevel
    };
  }

  /**
   * Automatically calculate student level based on Total XP
   */
  private calculateLevel(totalXP: number): any {
    // Return the highest level where requiredXP <= totalXP
    return LEVEL_THRESHOLDS.slice().reverse().find(tier => totalXP >= tier.requiredXP) || LEVEL_THRESHOLDS[0];
  }
}


// 3. MATH KINGDOM JOURNEY MAP SERVICE
export class JourneyMapService {
  private journeyRepo = new JourneyRepository();
  private xpService = new XpAndLevelService();
  private notificationService = new NotificationService();

  /**
   * Jab bacha Map mein ek Stage/Boss clear kare
   */
  public async completeJourneyStage(studentId: string, worldId: string, stageId: number, isBossStage: boolean): Promise<any> {
    
    // 1. Fetch current progress
    const progress = await this.journeyRepo.findProgressByStudent(studentId);
    
    let nextWorldUnlocked = false;
    let earnedXP = isBossStage ? 100 : 20; // Boss stage ka XP zyada hota hai

    // 2. XP Award Karo (Yahan upar wala XP engine kaam aayega)
    const xpResult = await this.xpService.awardXP(studentId, earnedXP, `Completed ${worldId} - Stage ${stageId}`);

    // 3. Agar Boss Stage thi, toh Agla World (Region) Unlock karo
    if (isBossStage) {
      progress.completedBosses.push(`${worldId}_BOSS`);
      
      // Auto-unlock logic for the next region (e.g., from Number Forest to Algebra Castle)
      const nextWorldId = this.getNextWorldId(worldId);
      if (nextWorldId) {
        progress.unlockedWorlds.push(nextWorldId);
        progress.currentWorld = nextWorldId;
        progress.currentStage = 1; // Start at stage 1 of new world
        nextWorldUnlocked = true;

        await this.notificationService.sendSystemNotification(
          studentId,
          "🏰 New World Unlocked!",
          `You have defeated the boss and entered ${nextWorldId}.`
        );
      }
    } else {
      progress.currentStage += 1; // Move to next stage in same world
    }

    // 4. Update the journey progress in Database
    await this.journeyRepo.updateProgress(studentId, progress);

    return {
      success: true,
      stageCompleted: stageId,
      earnedXP,
      levelUpData: xpResult,
      nextWorldUnlocked,
      currentStatus: progress
    };
  }

  private getNextWorldId(currentWorldId: string): string | null {
    const worldOrder = ['Home Village', 'Number Forest', 'Fraction River', 'Algebra Castle', 'Logic Cave', 'Geometry Temple'];
    const currentIndex = worldOrder.indexOf(currentWorldId);
    if (currentIndex !== -1 && currentIndex < worldOrder.length - 1) {
      return worldOrder[currentIndex + 1];
    }
    return null; // All worlds completed!
  }
}

/* =========================================================
   STEP 7: LEADERBOARD, STREAKS & WINNERS AUTOMATION
   ========================================================= */

import { StudentRepository } from '../repositories/student.repository';
import { AttemptRepository } from '../repositories/attempt.repository';
import { LeaderboardRepository } from '../repositories/leaderboard.repository';
import { NotificationService } from './notification.service';
import { BadgeService } from './badge.service';

// 1. STREAK MANAGEMENT SERVICE (Baccho ki consistency track karna)
export class StreakService {
  private studentRepo = new StudentRepository();
  private badgeService = new BadgeService();
  private notificationService = new NotificationService();

  /**
   * Jab bhi bacha Daily Challenge submit karega, yeh function call hoga
   */
  public async updateDailyStreak(studentId: string): Promise<any> {
    const student = await this.studentRepo.findById(studentId);
    if (!student) throw new Error("Student not found");

    let currentStreak = student.metrics.dailyStreak;
    const lastAttemptDate = await this.studentRepo.getLastAttemptDate(studentId);
    
    const today = new Date();
    const yesterday = new Date(today);
    yesterday.setDate(yesterday.getDate() - 1);

    // Agar bacche ne kal challenge diya tha, toh streak +1 karo
    if (this.isSameDate(lastAttemptDate, yesterday)) {
      currentStreak += 1;
    } 
    // Agar bacha aaj pehli baar aaya hai, ya beech mein din miss kar diya, toh streak 1 se shuru
    else if (!this.isSameDate(lastAttemptDate, today)) {
      currentStreak = 1;
    }

    // Best Streak Update
    const bestStreak = Math.max(currentStreak, student.metrics.bestStreak);

    // Save to Database
    await this.studentRepo.updateMetrics(studentId, {
      dailyStreak: currentStreak,
      bestStreak: bestStreak
    });

    // Check Milestones for Streak Badges (7 Days, 15 Days, 30 Days)
    await this.checkStreakMilestones(studentId, currentStreak);

    return { currentStreak, bestStreak };
  }

  private async checkStreakMilestones(studentId: string, streak: number) {
    if (streak === 7) {
      await this.badgeService.awardBadge(studentId, "7 Day Bronze Streak");
    } else if (streak === 15) {
      await this.badgeService.awardBadge(studentId, "15 Day Silver Streak");
    } else if (streak === 30) {
      await this.badgeService.awardBadge(studentId, "30 Day Gold Streak");
    }
  }

  private isSameDate(date1: Date, date2: Date): boolean {
    if (!date1 || !date2) return false;
    return date1.toISOString().split('T')[0] === date2.toISOString().split('T')[0];
  }
}


// 2. LEADERBOARD & WINNERS SERVICE (Top performers nikalna)
export class LeaderboardService {
  private attemptRepo = new AttemptRepository();
  private leaderboardRepo = new LeaderboardRepository();

  /**
   * Yeh function Midnight Automation (Cron Job) ke through call hoga
   * Aaj ke saare attempts dekhega aur Top 3 baccho ko select karega
   */
  public async generateDailyWinners(): Promise<void> {
    const today = new Date().toISOString().split('T')[0];
    
    // 1. Fetch all CORRECT attempts of today, sorted by 'timeTakenSeconds' (Fastest first)

        xpEarned: attempt.result.xpEarned,

      };

    // 3. Save to 'daily_winners' collection (Frontend will read directly from here)

  public async getOverallStatistics(): Promise<any> {
    const today = new Date().toISOString().split('T')[0];

    // Running parallel queries for speed (Performance optimization)
    const [totalStudents, newRegToday, totalAttempts, todayAttempts] = await Promise.all([
      this.studentRepo.countAll(),
      this.studentRepo.countRegisteredOnDate(today),
      this.attemptRepo.countAll(),
      this.attemptRepo.countAttemptsOnDate(today)
    ]);

    return {

      newRegistrationsToday: newRegToday,
      totalChallengesAttempted: totalAttempts,
      activeStudentsToday: todayAttempts
    };
  }

  /**
   * Questions ki report - kaunsa question hard tha, kaunsa easy
   */
  public async getQuestionAnalytics(questionId: string): Promise<any> {
    const attempts = await this.attemptRepo.findAttemptsByQuestionId(questionId);
    
    if (attempts.length === 0) return { message: "No attempts found for this question." };

    const totalAttempts = attempts.length;
    const correctAttempts = attempts.filter(a => a.result.isCorrect).length;
    
    let totalTime = 0;
    attempts.forEach(a => totalTime += a.timing.timeTakenSeconds);

    const accuracy = ((correctAttempts / totalAttempts) * 100).toFixed(2);
    const avgTime = (totalTime / totalAttempts).toFixed(2);

    return {
      questionId: questionId,
      totalAttempts: totalAttempts,
      accuracyPercentage: `${accuracy}%`,
      averageSolvingTimeSeconds: avgTime,
      difficultyRating: accuracy < 30 ? "Hard" : accuracy > 75 ? "Easy" : "Medium"
    };
  }
}

import express from 'express';
import rateLimit from 'express-rate-limit';
import helmet from 'helmet';
import cors from 'cors';

export class SecurityConfig {
  
  /**
   * Yeh function hamari Express App par saari security layers lagayega
   */
  public static applySecurity(app: express.Application) {
    
    // 1. HELMET: Hides server details and adds secure HTTP headers
    // Isse hackers ko pata nahi chalega ki server Node.js par hai ya kis version par hai.
    app.use(helmet());

    // 2. CORS (Cross-Origin Resource Sharing): 
    // Sirf aapki official website (e.g., amitolympiad.com) hi is API ko call kar sakti hai.
    app.use(cors({
      origin: ['https://www.amitolympiad.com', 'https://admin.amitolympiad.com'],
      methods: ['GET', 'POST', 'PUT', 'DELETE'],
      credentials: true
    }));

    // 3. GLOBAL RATE LIMITER (DDoS Protection)
    // Ek IP address 15 minute mein maximum 100 request hi bhej sakta hai.
    // Agar koi bot website ko hang karne ki koshish karega, toh system use block kar dega.
    const globalLimiter = rateLimit({
      windowMs: 15 * 60 * 1000, // 15 minutes
      max: 100, 
      message: { success: false, error: "⚠️ Too many requests from this IP, please try again after 15 minutes." },
      standardHeaders: true,
      legacyHeaders: false,
    });
    app.use('/api/', globalLimiter);

    // 4. OTP & LOGIN BRUTE FORCE PROTECTION (Strict Limiter)
    // OTP wale page par sirf 5 try allow honge, taaki koi OTP guess na kar sake.
    const strictAuthLimiter = rateLimit({
      windowMs: 10 * 60 * 1000, // 10 minutes
      max: 5, // Only 5 OTP requests allowed per IP
      message: { success: false, error: "🚨 Too many OTP attempts. Your IP has been temporarily blocked." }
    });
    app.use('/api/auth/send-otp', strictAuthLimiter);
    app.use('/api/auth/verify-otp', strictAuthLimiter);

    // 5. INPUT SANITIZATION & XSS PROTECTION
    // (We use express.json middleware with a size limit so large files don't crash the server)
    app.use(express.json({ limit: '10kb' })); 
  }
}

    // Yahan hum response mein 'FlaggedForReview' mark kar denge
    req.body.isSuspicious = true; 
    req.body.suspicionReason = 'High tab switch count';
  }

  // Rule 2: Agar kisi bache ne 1 second mein hi math ka hard question solve kar diya (Robot/Script)
  if (timeTakenSeconds < 2) {
    console.warn(`🚩 Anti-Cheat Alert: User ${req.user?.userId} answered unnaturally fast (${timeTakenSeconds}s)!`);
    req.body.isSuspicious = true;
    req.body.suspicionReason = 'Impossible completion time';
  }

  next();
};

// ---------------------------------------------------------
// 3. SCALABILITY & CLUSTERING (Feature 19)
// ---------------------------------------------------------
/**
 * Yeh server.ts ka main entry point hai.
 * Normal Node.js sirf 1 CPU core use karta hai. 
 * Yeh logic server ke saare CPU cores (e.g. 8 cores) ko use karega taaki 8 guna zyada load utha sake!
 */
export const startScalableServer = (app: express.Application, port: number | string) => {
  const numCPUs = os.cpus().length;

  if (cluster.isPrimary) {
    logger.info(`👑 Master Server is running on PID: ${process.pid}`);
    logger.info(`🚀 Forking ${numCPUs} Worker Nodes for maximum performance...`);

    // Har CPU core ke liye ek naya 'Worker' (server clone) banao
    for (let i = 0; i < numCPUs; i++) {
      cluster.fork();
    }

    // Agar koi worker (server clone) crash ho jaye, toh turant naya chalu kar do (Auto-Healing)
    cluster.on('exit', (worker, code, signal) => {
      logger.error(`❌ Worker ${worker.process.pid} died. Restarting a new worker...`);
      cluster.fork(); 
    });

  } else {
    // Worker nodes par actual express server chalega
    app.listen(port, () => {
      logger.info(`⚡ Worker Node (PID: ${process.pid}) started and listening on port ${port}`);
    });
  }
};

  public async createOrder(studentId: string, courseId: string, amountINR: number): Promise<any> {
    
    const options = {
      amount: amountINR * 100, // Razorpay paise (paisa) mein amount leta hai, isliye 100 se multiply
      currency: "INR",
      receipt: `receipt_${studentId}_${Date.now()}`,
      payment_capture: 1 // Auto-capture payment

      // Save pending order in Database
      // await this.orderRepo.createPendingOrder(studentId, courseId, order.id);

      return {
        success: true,
        orderId: order.id,
        amount: order.amount,
        currency: order.currency
      };
    } catch (error) {
      throw new Error("Payment initialization failed.");
    }
  }

  /**
   * STEP B: Webhook/Signature Verification (Payment successful hone ke baad)
   * Hackers fake 'success' message bhej sakte hain, yeh function check karega ki paisa sach me aaya ya nahi.
   */
  public async verifyPayment(razorpayOrderId: string, razorpayPaymentId: string, signature: string, studentId: string, courseId: string): Promise<any> {
    
    // Server apni secret key se signature banayega aur check karega
    const generatedSignature = crypto
      .createHmac('sha256', process.env.RAZORPAY_KEY_SECRET!)
      .update(razorpayOrderId + "|" + razorpayPaymentId)
      .digest('hex');

    if (generatedSignature !== signature) {
      // Fake Payment Attempt
      await this.auditLogger.logAction(studentId, 'STUDENT', 'FAKE_PAYMENT_ATTEMPT', `Failed signature match for order ${razorpayOrderId}`, 'Server');
      throw new Error("❌ Payment Verification Failed! Fraud detected.");
    }

    // Payment Genuine Hai! Ab bache ko Premium bana do
    await this.studentRepo.upgradeToPremium(studentId, courseId);

    // Send Success Notification & Bill (Invoice)
    await this.notificationService.sendSystemNotification(
      studentId,
      "🎉 Premium Unlocked!",
      "Aapki payment successful ho gayi hai aur aapka Olympiad Video Course unlock ho gaya hai.",
      'High'
    );

    return {
      success: true,
      message: "Payment successful and course unlocked."
    };
  }
}

import { Server, Socket } from 'socket.io';
import { Server as HttpServer } from 'http';
import jwt from 'jsonwebtoken';
import { logger } from '../utils/logger.util';

export class ChatWebSocketService {
  private io: Server;

  /**
   * Main Express server (HTTP) ke upar hum Socket (Live) server lagayenge
   */
  public initializeSockets(httpServer: HttpServer) {
    this.io = new Server(httpServer, {
      cors: {
        origin: ['https://www.amitolympiad.com', 'https://admin.amitolympiad.com'],
        methods: ['GET', 'POST']
      }
    });

    // 1. SOCKET AUTHENTICATION (Sirf asli users chat kar payein)
    this.io.use((socket, next) => {
      const token = socket.handshake.auth.token;
      if (!token) {
        return next(new Error("Authentication Error: Token missing"));
      }
      try {
        const decoded = jwt.verify(token, process.env.JWT_SECRET_KEY!);
        // User ka data socket mein save kar lo
        (socket as any).user = decoded; 
        next();
      } catch (err) {
        return next(new Error("Authentication Error: Invalid token"));
      }
    });

    // 2. LISTEN FOR CONNECTIONS
    this.io.on('connection', (socket: Socket) => {
      const user = (socket as any).user;
      logger.info(`💬 User connected to Live Chat: ${user.userId}`);

      // Bacche ko uske personal chat room mein daal do
      socket.join(`room_${user.userId}`);
      
      // Agar Admin login karta hai, toh use 'admin_room' mein daal do
      if (user.role === 'ADMIN' || user.role === 'SUPER_ADMIN') {
        socket.join('admin_room');
      }

      // 3. LISTEN FOR MESSAGES FROM STUDENT
      socket.on('student_send_message', async (data) => {
        const { text } = data;
        
        // A. Pehle check karo agar Bot jawab de sakta hai (FAQ)
        const botReply = this.getAutoBotReply(text);
        
        if (botReply) {
          // Instant reply back to student
          socket.emit('receive_message', {
            sender: 'Bot',
            text: botReply,
            timestamp: new Date()
          });
        } else {
          // B. Agar Bot ko nahi pata, toh message Admin Dashboard par bhej do
          socket.emit('receive_message', {
            sender: 'Bot',
            text: "Main aapka message Admin sir ko forward kar raha hoon. Kripya thodi der wait karein.",
            timestamp: new Date()
          });

          this.io.to('admin_room').emit('admin_receive_query', {
            studentId: user.userId,
            text: text,
            timestamp: new Date()
          });
        }
      });

      // 4. LISTEN FOR ADMIN REPLY
      socket.on('admin_send_reply', (data) => {
        const { studentId, text } = data;
        
        // Admin ka reply seedha us specific bacche ke room mein bhej do
        this.io.to(`room_${studentId}`).emit('receive_message', {
          sender: 'Admin',
          text: text,
          timestamp: new Date()
        });
      });

      socket.on('disconnect', () => {
        logger.info(`🔴 User disconnected from chat: ${user.userId}`);
      });
    });
  }

  /**
   * Simple FAQ Bot Logic (Isme aap AI ChatGPT bhi connect kar sakte hain future me)
   */
  private getAutoBotReply(message: string): string | null {
    const msg = message.toLowerCase();
    
    if (msg.includes("password") || msg.includes("login")) {
      return "Agar aap password bhool gaye hain, toh login page par 'Forgot Password' par click karein aur apna mobile number dalein.";
    }
    if (msg.includes("certificate") || msg.includes("download")) {
      return "Aapka certificate aapke 'Profile > My Achievements' section mein available hai. Wahan se PDF download kar lijiye.";
    }
    if (msg.includes("fees") || msg.includes("payment")) {
      return "Olympiad ki registration fees aur courses ki detail 'Pricing' page par di gayi hai. Aap UPI, Paytm ya Card se pay kar sakte hain.";
    }
    
    // Return null agar message normal FAQ se bahar hai
    return null;
  }
}

import express, { Request, Response } from 'express';
import { AttemptRepository } from '../repositories/attempt.repository';
import { StreakService } from './streak.service';
import { logger } from '../utils/logger.util';

export class OfflineSyncService {
  private attemptRepo = new AttemptRepository();
  private streakService = new StreakService();

  /**
   * STEP A: Background Sync API (Jab net wapas aaye tab yeh call hogi)
   * Bacha offline tha, usne 5 sawal kiye, ab net aane par frontend ek sath 5 answers yahan bhejega.
   */
  public async syncOfflineData(studentId: string, offlineAttempts: any[]): Promise<any> {
    logger.info(`🔄 Starting offline data sync for Student: ${studentId}. Total items: ${offlineAttempts.length}`);
    
    let successCount = 0;
    let failCount = 0;

    // Har ek answer ko database mein process karna
    for (const attempt of offlineAttempts) {
      try {
        // Check if this attempt is already synced (to prevent duplicate entries)
        const isDuplicate = await this.attemptRepo.checkIfExists(attempt.localAttemptId);
        
        if (!isDuplicate) {
          // Process and save the attempt
          await this.attemptRepo.saveAttempt({
            studentId: studentId,
            questionId: attempt.questionId,
            selectedAnswer: attempt.selectedAnswer,
            timeTakenSeconds: attempt.timeTakenSeconds,
            isCorrect: attempt.isCorrect,
            submittedAt: attempt.offlineTimestamp // Bacha jis time offline tha, wahi original time save hoga
          });
          
          successCount++;
        }
      } catch (error) {
        logger.error(`❌ Failed to sync attempt ${attempt.localAttemptId}`, error);
        failCount++;
      }
    }

    // Ek baar saara offline data sync ho jaye, toh bacche ki Daily Streak update kar do
    if (successCount > 0) {
      await this.streakService.updateDailyStreak(studentId);
    }

    return {
      success: true,
      message: `Offline sync complete. Saved: ${successCount}, Failed: ${failCount}`
    };
  }
}
