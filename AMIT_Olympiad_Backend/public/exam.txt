    <script>
        // 1. OPTION SELECTION LOGIC
        function selectOption(element) {
            document.querySelectorAll('.option-label').forEach(el => el.classList.remove('selected'));
            element.classList.add('selected');
            element.querySelector('input').checked = true;
        }

        // 2. LIVE TIMER LOGIC
        let totalSeconds = 3600; // 60 minutes
        setInterval(() => {
            if (totalSeconds <= 0) { submitExam(); return; }
            totalSeconds--;
            let m = Math.floor(totalSeconds / 60);
            let s = totalSeconds % 60;
            document.getElementById("timeRemaining").innerText = `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
        }, 1000);

        // 3. ANTI-CHEATING (TAB SWITCH DETECTION)
        let warnings = 0;
        document.addEventListener("visibilitychange", () => {
            if (document.visibilityState === "hidden") {
                warnings++;
                if(warnings > 2) {
                    alert("Exam Auto-Submitted due to multiple tab switches.");
                    submitExam();
                } else {
                    document.getElementById("warningOverlay").style.display = "flex";
                }
            }
        });

        function dismissWarning() {
            document.getElementById("warningOverlay").style.display = "none";
            requestFullScreen();
        }

        // 4. FULL SCREEN LOCK
        function requestFullScreen() {
            let elem = document.documentElement;
            if (elem.requestFullscreen) { elem.requestFullscreen(); }
        }

        // 5. BUTTON SIMULATIONS
        function saveAndNext() {
            document.querySelector('.q-box.active-q').className = "q-box answered";
            alert("Answer Saved in Database. Loading next question...");
        }

        function markForReview() {
            document.querySelector('.q-box.active-q').className = "q-box marked";
        }

        function submitExam() {
            alert("🎯 Exam Submitted Successfully! Redirecting to Result Portal...");
            window.location.href = "index.html"; // Will change to result portal later
        }

        // Disable right click and copy to prevent cheating
        document.addEventListener('contextmenu', event => event.preventDefault());
        document.addEventListener('copy', event => {
            event.preventDefault();
            alert("Copying is strictly prohibited during the exam!");
        });
    </script>
